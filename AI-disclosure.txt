Google AI was used to recap how to write structured data as JSON-LD

To structure data using JSON-LD (JavaScript Object Notation for Linked Data), you must embed a standard JSON object inside an isolated <script type="application/ld+json"> tag. Google officially recommends this format for defining web page semantics.
Below is a template showing a unified @graph structure—the highly optimized format preferred by modern search engines and AI crawlers to understand entity relationships on a single page.
html
<script type="application/ld+json">
{
  "@context": "https://schema.org",
  "@graph": [
    {
      "@type": "WebSite",
      "@id": "https://example.com",
      "url": "https://example.com",
      "name": "Example Brand",
      "description": "An example website showcasing structured data.",
      "publisher": {
        "@id": "https://example.com#organization"
      }
    },
    {
      "@type": "Organization",
      "@id": "https://example.com#organization",
      "name": "Example Corp",
      "url": "https://example.com",
      "logo": {
        "@type": "ImageObject",
        "url": "https://example.comlogo.png"
      },
      "sameAs": [
        "https://facebook.com",
        "https://twitter.com"
      ]
    },
    {
      "@type": "WebPage",
      "@id": "https://example.compage/#webpage",
      "url": "https://example.compage/",
      "name": "Example Page Title",
      "isPartOf": {
        "@id": "https://example.com"
      },
      "about": {
        "@id": "https://example.com#organization"
      }
    }
  ]
}
</script>
